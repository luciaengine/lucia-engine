cmake_minimum_required(VERSION 3.25)
project(RuntimeDependencyStagingProbe LANGUAGES C)

if(APPLE)
    set(CMAKE_INSTALL_RPATH "@loader_path")
    set(CMAKE_INSTALL_NAME_DIR "@rpath")
elseif(UNIX)
    set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/leaf.c" "int leaf(void) { return 7; }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/child.c" "int leaf(void); int child(void) { return leaf(); }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/oldchild.c" "int retired(void); int child(void) { return retired(); }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/retired.c" "int retired(void) { return 3; }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/parent.c" "int child(void); int parent(void) { return child(); }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/host.c" "int parent(void); int main(void) { return parent(); }\n")
add_library(Leaf SHARED "${CMAKE_CURRENT_BINARY_DIR}/leaf.c")
add_library(Child SHARED "${CMAKE_CURRENT_BINARY_DIR}/child.c")
add_library(Parent SHARED "${CMAKE_CURRENT_BINARY_DIR}/parent.c")
target_link_libraries(Child PRIVATE Leaf)
target_link_libraries(Parent PRIVATE Child)
add_library(OldChild SHARED "${CMAKE_CURRENT_BINARY_DIR}/oldchild.c")
add_library(Retired SHARED "${CMAKE_CURRENT_BINARY_DIR}/retired.c")
target_link_libraries(OldChild PRIVATE Retired)
add_library(OldParent SHARED "${CMAKE_CURRENT_BINARY_DIR}/parent.c")
target_link_libraries(OldParent PRIVATE OldChild)
set_target_properties(OldChild PROPERTIES OUTPUT_NAME Child)
set_target_properties(OldParent PROPERTIES OUTPUT_NAME Parent)
foreach(_target Leaf Child Parent)
    set_target_properties(${_target} PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fresh/$<CONFIG>"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fresh/$<CONFIG>"
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fresh/$<CONFIG>")
endforeach()
foreach(_target OldChild OldParent Retired)
    set_target_properties(${_target} PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/old/$<CONFIG>"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/old/$<CONFIG>"
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/old/$<CONFIG>")
endforeach()
add_executable(Host "${CMAKE_CURRENT_BINARY_DIR}/host.c")
target_link_libraries(Host PRIVATE OldParent)
set_target_properties(Host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/stage/$<CONFIG>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/paths_$<CONFIG>.cmake" CONTENT
"set(HOST [==[$<TARGET_FILE:Host>]==])
set(FRESH [==[$<TARGET_FILE_DIR:Parent>]==])
set(OLD_PARENT [==[$<TARGET_FILE:OldParent>]==])
set(OLD_CHILD [==[$<TARGET_FILE:OldChild>]==])
set(RETIRED [==[$<TARGET_FILE:Retired>]==])
set(PARENT [==[$<TARGET_FILE_NAME:Parent>]==])
set(CHILD [==[$<TARGET_FILE_NAME:Child>]==])
set(LEAF [==[$<TARGET_FILE_NAME:Leaf>]==])
set(UNRELATED [==[${CMAKE_SHARED_LIBRARY_PREFIX}Unrelated${CMAKE_SHARED_LIBRARY_SUFFIX}]==])
")
