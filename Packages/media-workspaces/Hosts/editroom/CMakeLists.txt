cmake_minimum_required(VERSION 3.25)
project(LuciaPortableEditroom LANGUAGES CXX)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets)
add_library(LuciaMediaWorkspace SHARED LuciaEditroom.cpp RoomDocument.cpp ../QtWorkspaceHost.cpp)
target_compile_features(LuciaMediaWorkspace PRIVATE cxx_std_20)
target_link_libraries(LuciaMediaWorkspace PRIVATE Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
if(WIN32)
    target_compile_options(LuciaMediaWorkspace PRIVATE /W4 /WX /utf-8)
    target_link_libraries(LuciaMediaWorkspace PRIVATE user32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(X11 REQUIRED)
    target_compile_options(LuciaMediaWorkspace PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(LuciaMediaWorkspace PRIVATE X11::X11)
else()
    message(FATAL_ERROR "Build the native Editroom workspace on macOS.")
endif()
install(TARGETS LuciaMediaWorkspace RUNTIME DESTINATION Editroom LIBRARY DESTINATION Editroom)
install(FILES Editroom-LICENSE DESTINATION Editroom)

include(CTest)
if(BUILD_TESTING)
    add_executable(RoomDocumentTests Tests/RoomDocumentTests.cpp RoomDocument.cpp)
    target_include_directories(RoomDocumentTests PRIVATE .)
    target_compile_features(RoomDocumentTests PRIVATE cxx_std_20)
    target_link_libraries(RoomDocumentTests PRIVATE Qt6::Core)
    add_test(NAME RoomDocumentTests COMMAND RoomDocumentTests)
endif()
